// makeMarkerDataUri 안의 베이스 도형(현재 <circle cx=50 cy=44 r=40 .../>)을
// 아래 affilFrame(type) 호출로 교체.
// 앱 viewBox36 → 웹(중심 50,44, 반경~40) 변환 적용한 path.

// [추가] 소속별 프레임 SVG (베이스). 앱 모양과 동일.
function affilFrame(type) {
  const color = AFFIL_COLORS[type] || '#888';
  const st = 'stroke="#0A1628" stroke-width="4"';
  switch (type) {
    case 0: // HOSTILE 마름모
      return `<path d="M50,7 L83,44 L50,81 L17,44 Z" fill="${color}" ${st}/>`;
    case 1: // FRIENDLY 원
      return `<circle cx="50" cy="44" r="37" fill="${color}" ${st}/>`;
    case 2: // UNKNOWN 사각
      return `<rect x="15" y="9" width="70" height="70" fill="${color}" ${st}/>`;
    case 3: // NEUTRAL 사각
      return `<rect x="15" y="9" width="70" height="70" fill="${color}" ${st}/>`;
    case 4: // POI 별
      return `<path d="M50,5 L61,33 L91,33 L67,52 L75,82 L50,63 L25,82 L33,52 L9,33 L39,33 Z" fill="${color}" stroke="#0A1628" stroke-width="3"/>`;
    case 5: // ENGAGED 이중마름모
      return `<path d="M50,2 L86,44 L50,86 L14,44 Z" fill="none" stroke="#FF9800" stroke-width="6"/>`
           + `<path d="M50,14 L78,44 L50,74 L22,44 Z" fill="#E53935" stroke="#0A1628" stroke-width="3"/>`;
    case 6: // THREAT 삼각
      return `<path d="M50,7 L85,79 L15,79 Z" fill="${color}" ${st}/>`;
    default:
      return `<circle cx="50" cy="44" r="37" fill="${color}" ${st}/>`;
  }
}

// makeMarkerDataUri 내부 수정:
//   기존: <circle cx="50" cy="44" r="40" fill="${color}" stroke="#0A1628" stroke-width="4"/>
//   교체: ${affilFrame(type)}
// (color 변수는 affilFrame 안에서 쓰므로 바깥 const color 줄은 둬도 무방)
